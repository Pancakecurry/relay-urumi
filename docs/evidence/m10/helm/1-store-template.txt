apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: release-check-db
spec:
  serviceName: release-check-db
  replicas: 1
  selector:
    matchLabels: {app: "release-check-db"}
  template:
    metadata:
      labels: {app: "release-check-db", role: database}
    spec:
      automountServiceAccountToken: false
      securityContext:
        seccompProfile: {type: RuntimeDefault}
      containers:
        - name: mariadb
          image: "mariadb:11.8.9-noble@sha256:6422478cb8e159f080fb1d8ccf65101e26fe51385787fde7d16c3b165a331f15"
          args:
            - --innodb-buffer-pool-size=128M
            - --max-connections=40
          env:
            - {name: MARIADB_DATABASE, value: wordpress}
            - {name: MARIADB_USER, value: wordpress}
            - {name: MARIADB_PASSWORD_FILE, value: /run/store/db-password}
            - {name: MARIADB_ROOT_PASSWORD_FILE, value: /run/store/root-password}
          ports: [{name: mysql, containerPort: 3306}]
          startupProbe:
            exec: {command: [healthcheck.sh, --connect, --innodb_initialized]}
            periodSeconds: 5
            failureThreshold: 60
          readinessProbe:
            exec: {command: [healthcheck.sh, --connect, --innodb_initialized]}
            periodSeconds: 10
            timeoutSeconds: 5
          livenessProbe:
            exec: {command: [healthcheck.sh, --connect, --innodb_initialized]}
            periodSeconds: 20
            timeoutSeconds: 5
            failureThreshold: 6
          resources: 
            limits:
              cpu: 750m
              memory: 512Mi
            requests:
              cpu: 100m
              memory: 128Mi
          volumeMounts:
            - {name: data, mountPath: /var/lib/mysql}
            - {name: runtime, mountPath: /run/store, readOnly: true}
      volumes:
        - name: data
          persistentVolumeClaim: {claimName: "release-check-db"}
        - name: runtime
          secret:
            secretName: store-runtime
            defaultMode: 0440
            items:
              - {key: db-password, path: db-password}
              - {key: root-password, path: root-password}
